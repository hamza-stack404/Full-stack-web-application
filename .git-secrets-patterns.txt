# Git secrets patterns to prevent API key leaks
# Add this to pre-commit hooks

# Generic API key patterns
[aA][pP][iI]_?[kK][eE][yY].*['"][0-9a-zA-Z]{32,45}['"]

# AWS Keys
AKIA[0-9A-Z]{16}
